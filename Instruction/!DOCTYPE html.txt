function getArray(response){

    if(!response){
        return [];
    }

    if(Array.isArray(response)){
        return response;
    }

    if(Array.isArray(response.value)){
        return response.value;
    }

    if(response.data && Array.isArray(response.data.value)){
        return response.data.value;
    }

    if(response.body && Array.isArray(response.body.value)){
        return response.body.value;
    }

    if(typeof response === "string"){

        try{

            const parsed =
            JSON.parse(response);

            if(Array.isArray(parsed)){
                return parsed;
            }

            if(Array.isArray(parsed.value)){
                return parsed.value;
            }

            if(parsed.data && Array.isArray(parsed.data.value)){
                return parsed.data.value;
            }

            if(parsed.body && Array.isArray(parsed.body.value)){
                return parsed.body.value;
            }

        }catch(error){
            return [];
        }

    }

    return [];

}

function today(){

    return new Date().toISOString().split("T")[0];

}

function safeNumber(value){

    const num =
    Number(value);

    return isNaN(num) ? 0 : num;

}

function escapeHtml(value){

    return String(value || "")
        .replace(/&/g,"&amp;")
        .replace(/</g,"&lt;")
        .replace(/>/g,"&gt;")
        .replace(/"/g,"&quot;")
        .replace(/'/g,"&#039;");

}

function selectedCustomer(){

    const code =
    customerSelect.value;

    return customers.find(customer =>
        String(customer.CardCode) === String(code)
    );

}

function getItemByCode(itemCode){

    return items.find(item =>
        String(item.ItemCode) === String(itemCode)
    );

}

function isBatchManaged(item){

    return item &&
    (
        item.ManageBatchNumbers === "tYES" ||
        item.ManageBatchNumbers === true
    );

}

function isSerialManaged(item){

    return item &&
    (
        item.ManageSerialNumbers === "tYES" ||
        item.ManageSerialNumbers === true
    );

}

function isInventoryItem(item){

    return item &&
    (
        item.InventoryItem === "tYES" ||
        item.InventoryItem === true
    );

}

function loadCustomers(){

    customerSelect.innerHTML =
    `<option value="">-- Select Customer --</option>`;

    customers.forEach(customer => {

        const option =
        document.createElement("option");

        option.value =
        customer.CardCode;

        option.textContent =
        `${customer.CardCode} - ${customer.CardName || ""}`;

        customerSelect.appendChild(option);

    });

}

function buildItemOptions(){

    let html = `<option value="">-- Select Item --</option>`;

    items.forEach(item => {
        // Fallbacks in case SAP properties are missing
        const itemCode = item.ItemCode || "";
        const itemName = item.ItemName || "";

        const batchBadge = isBatchManaged(item) ? " | Batch" : "";
        const serialBadge = isSerialManaged(item) ? " | Serial" : "";
        const inventoryBadge = isInventoryItem(item) ? " | Inventory" : " | Non-Inventory";

        html += `<option value="${escapeHtml(itemCode)}">
            ${escapeHtml(itemCode)} - ${escapeHtml(itemName)}${batchBadge}${serialBadge}${inventoryBadge}
        </option>`;

    });

    return html;

}

function buildWarehouseOptions(){

    let html =
    `<option value="">-- Select Warehouse --</option>`;

    warehouses.forEach(warehouse => {

        html +=
        `<option value="${escapeHtml(warehouse.WarehouseCode)}">
            ${escapeHtml(warehouse.WarehouseCode)} - ${escapeHtml(warehouse.WarehouseName)}
        </option>`;

    });

    return html;

}

    document.getElementById(`type_${index}`);

function getAvailableBatches(itemCode){

    return batches.filter(batch => {

        const sameItem =
        String(batch.ItemCode) === String(itemCode);

        const notLocked =
        batch.Status !== "bdsStatus_Locked";

        return sameItem && notLocked;

    });

}

function getAvailableSerials(itemCode){

    return serials.filter(serial => {

        return String(serial.ItemCode) === String(itemCode);

    });

}

function renderAllocationSection(index){

    const itemCode =
    document.getElementById(`item_${index}`).value;

    const item =
    getItemByCode(itemCode);

    const allocationBox =
    document.getElementById(`allocation_${index}`);

    if(!item){

        allocationBox.innerHTML =
        "No allocation required yet.";

        return;

    }

    if(isBatchManaged(item)){

        allocationBox.innerHTML =
        `
            <div class="batch-box">
                <b>Batch Allocation</b>
                <div id="batchRows_${index}"></div>
                <button type="button" class="primary" onclick="addBatchRow(${index})">+ Add Batch</button>
                <div class="small">Batch quantity total must equal line quantity.</div>
            </div>
        `;

        addBatchRow(index);

        return;

    }

    if(isSerialManaged(item)){

        const availableSerials =
        getAvailableSerials(itemCode);

        let options =
        "";

        availableSerials.forEach(serial => {

            const serialNo =
            serial.SerialNumber || "";

            const systemNo =
            serial.SystemNumber || "";

            if(!serialNo){
                return;
            }

            const value =
            escapeHtml(JSON.stringify({
                SerialNumber:
                serialNo,

                SystemNumber:
                systemNo
            }));

            options +=
            `<option value="${value}">
                ${escapeHtml(serialNo)} | System No: ${escapeHtml(systemNo)}
            </option>`;

        });

        allocationBox.innerHTML =
        `
            <div class="serial-box">
                <b>Serial Allocation</b>
                <select id="serialSelect_${index}" multiple style="min-height:130px;">
                    ${options}
                </select>
                <div class="small">Selected serial count must equal line quantity.</div>
            </div>
        `;

        return;

    }

    allocationBox.innerHTML =
    `<span class="small">No batch/serial allocation required.</span>`;

}

function addBatchRow(index){

    const itemCode =
    document.getElementById(`item_${index}`).value;

    const availableBatches =
    getAvailableBatches(itemCode);

    const box =
    document.getElementById(`batchRows_${index}`);

    if(!box){
        return;
    }

    const row =
    document.createElement("div");

    row.className =
    "batch-row";

    let options =
    `<option value="">Select Batch</option>`;

    availableBatches.forEach(batch => {

        const batchNo =
        batch.Batch || "";

        if(!batchNo){
            return;
        }

        options +=
        `<option value="${escapeHtml(batchNo)}">
            ${escapeHtml(batchNo)} | System No: ${escapeHtml(batch.SystemNumber || "")} | ${escapeHtml(batch.Status || "")}
        </option>`;

    row.innerHTML =
    `
        <select class="batchSelect">
            ${options}
        </select>

        <input class="batchQty" type="number" min="0" step="1" placeholder="Qty">

        <button type="button" class="danger" onclick="this.parentElement.remove()">Remove</button>
    `;

    box.appendChild(row);

}

function collectBatchNumbers(index,lineQty){

    const box =
    document.getElementById(`batchRows_${index}`);

    if(!box){
        return [];
    }

    const rows =
    box.querySelectorAll(".batch-row");

    const result =
    [];

    let total =
    0;

    rows.forEach(row => {

        const batchNo =
        row.querySelector(".batchSelect").value;

        const qty =
        safeNumber(row.querySelector(".batchQty").value);

        if(batchNo && qty > 0){

            result.push({
                BatchNumber:
                batchNo,

                Quantity:
                qty
            });

            total +=
            qty;

        }

    });

    if(total !== lineQty){

        throw new Error(
            `Batch quantity mismatch on line ${Number(index) + 1}. Line quantity is ${lineQty}, but selected batch quantity is ${total}.`
        );

    }

    return result;

}

function collectSerialNumbers(index,lineQty){

    const serialSelect =
    document.getElementById(`serialSelect_${index}`);

    if(!serialSelect){
        return [];
    }

    const selected =
    Array.from(serialSelect.selectedOptions).map(option =>
        JSON.parse(option.value)
    );

    if(selected.length !== lineQty){

        throw new Error(
            `Serial count mismatch on line ${Number(index) + 1}. Quantity is ${lineQty}, but selected serial count is ${selected.length}.`
        );

    }

    return selected.map(serial => {

        return {
            InternalSerialNumber:
            serial.SerialNumber,

            SystemSerialNumber:
            Number(serial.SystemNumber)
        };

    });

}

function updateTotals(){

    count;

    document.getElementById("subtotal").innerText =
    subtotal.toFixed(2);

    document.getElementById("reasonSummary").innerText =
    document.getElementById("reason").value;

    updateAuditPreview();

}

function buildPayload(){

    const customer =
    selectedCustomer();

    if(!customer){
        throw new Error("Please select customer.");
    }

    const rows =
    lineBody.querySelectorAll("tr");

    if(rows.length === 0){
        throw new Error("Please add at least one debit memo line.");
    }

    const documentLines =
    [];

    rows.forEach(row => {

        const index =
        row.dataset.index;

        const itemCode =
        document.getElementById(`item_${index}`).value;

        const item =
        getItemByCode(itemCode);

        const desc =
        document.getElementById(`desc_${index}`).value.trim();

        const warehouse =
        document.getElementById(`whs_${index}`).value.trim();

        const qty =
        safeNumber(document.getElementById(`qty_${index}`).value);

        const price =
        safeNumber(document.getElementById(`price_${index}`).value);

        const taxCode =
        document.getElementById(`tax_${index}`).value.trim();

        if(!itemCode){
            throw new Error(`Please select item on line ${Number(index) + 1}.`);
        }

        if(qty <= 0){
            throw new Error(`Quantity must be greater than 0 on line ${Number(index) + 1}.`);
        }

        if(price < 0){
            throw new Error(`Unit price cannot be negative on line ${Number(index) + 1}.`);
        }

        const line =
        {
            ItemCode:
            itemCode,

            Quantity:
            qty,

            UnitPrice:
            price,

            FreeText:
            desc
        };

        if(taxCode){
            line.TaxCode =
            taxCode;
        }

        /*
          WarehouseCode is required for inventory items.
          For non-inventory/service items, we avoid sending warehouse.
        */
        if(isInventoryItem(item)){

            if(!warehouse){
                throw new Error(`Please select warehouse on line ${Number(index) + 1}.`);
            }

            line.WarehouseCode =
            warehouse;

        }

        if(isBatchManaged(item)){

            line.BatchNumbers =
            collectBatchNumbers(index,qty);

        }

        if(isSerialManaged(item)){

            line.SerialNumbers =
            collectSerialNumbers(index,qty);

        }

        documentLines.push(line);

    });

    const reason =
    document.getElementById("reason").value;

    const baseReference =
    document.getElementById("baseReference").value.trim();

    const justification =
    document.getElementById("justification").value.trim();

    const comments =
    `Debit Memo Reason: ${reason} | Base Ref: ${baseReference || "N/A"} | Justification: ${justification || "N/A"} | Created by Debit Memo Agent`;

    const payload =
    {
        CardCode:
        customer.CardCode,

        DocDate:
        document.getElementById("docDate").value,

        DocDueDate:
        document.getElementById("dueDate").value,

        TaxDate:
        document.getElementById("taxDate").value,

        NumAtCard:
        document.getElementById("numAtCard").value.trim(),

        Comments:
        comments,

        JournalMemo:
        `Debit Memo - ${reason}`,

        DocumentLines:
        documentLines
    };

    /*
      If your SAP has UDF U_AuditTrail, uncomment this.
      If SAP gives invalid property U_AuditTrail, keep it commented.
    */
    // payload.U_AuditTrail = auditPreview.innerText;

    return payload;

}

function previewPayload(){

    try{

        const payload =
        buildPayload();

        const debugBox =
        document.getElementById("debugBox");

        debugBox.style.display =
        "block";

        debugBox.innerText =
        JSON.stringify(payload,null,2);

    }catch(error){

        alert(error.message);

    }

}

function submitDebitMemo(){

    try{

        const payload =
        buildPayload();

        window.parent.postMessage({

            type:
            "ORDER_FORM_SUBMIT",

            payload:
            payload

        },"*");

        alert("Debit Memo payload sent to flow.");

    }catch(error){

        alert(error.message);

    }

}
